cmake_minimum_required(VERSION 3.20)
project(remap_input LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(ANDROID)
    add_library(remap_input SHARED native_input.cpp android_routing.cpp watchdog.cpp)
    target_compile_definitions(remap_input PRIVATE _GNU_SOURCE)
    target_compile_options(remap_input PRIVATE -Wall -Wextra -Werror)
    target_link_options(remap_input PRIVATE "-Wl,-z,max-page-size=16384")
    option(REMAP_BUILD_EMULATOR_PROBE "Enable standalone synthetic-input emulator probe" OFF)
    if(REMAP_BUILD_EMULATOR_PROBE)
        add_executable(evdev_emulator_probe EXCLUDE_FROM_ALL
            ../../test/cpp/emulator_probe.cpp android_routing.cpp watchdog.cpp)
        target_include_directories(evdev_emulator_probe PRIVATE .)
        target_compile_definitions(evdev_emulator_probe PRIVATE _GNU_SOURCE)
        target_compile_options(evdev_emulator_probe PRIVATE -Wall -Wextra -Werror)
        target_link_options(evdev_emulator_probe PRIVATE "-Wl,-z,max-page-size=16384")
    endif()
else()
    enable_testing()
    add_executable(input_tests ../../test/cpp/input_tests.cpp android_routing.cpp)
    target_include_directories(input_tests PRIVATE .)
    add_test(NAME input_safety COMMAND input_tests)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_executable(watchdog_tests ../../test/cpp/watchdog_tests.cpp watchdog.cpp)
        target_include_directories(watchdog_tests PRIVATE .)
        add_test(NAME frozen_helper_watchdog COMMAND watchdog_tests)
        set_tests_properties(frozen_helper_watchdog PROPERTIES TIMEOUT 15)
    endif()
endif()
